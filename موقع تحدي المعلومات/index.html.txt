  <script>
    let fullQuestionBank = [];
    let selectedQuestions = [];
    let currentQuestionIndex = 0;
    let totalScore = 0;
    let timeLeft = 10;
    let timerInterval = null;
    let canAnswer = true;

    const fullDash = 251.2;

    function shuffleArray(array) {
      let shuffled = [...array];
      for (let i = shuffled.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1));
        [shuffled[i], shuffled[j]] = [shuffled[j], shuffled[i]];
      }
      return shuffled;
    }

    async function loadQuestions() {
      try {
        const response = await fetch('questions.json');
        fullQuestionBank = await response.json();
        startNewGame();
      } catch (error) {
        document.getElementById('question-text').innerText = 'خطأ في تحميل ملف الأسئلة!';
      }
    }

    function startNewGame() {
      totalScore = 0;
      currentQuestionIndex = 0;
      canAnswer = true;
      selectedQuestions = shuffleArray(fullQuestionBank).slice(0, 10);

      document.getElementById('quiz-screen').style.display = 'block';
      document.getElementById('result-screen').style.display = 'none';
      document.getElementById('score-live').innerText = `النقاط: 0`;

      showQuestion();
    }

    function updateTimerUI() {
      document.getElementById('timer-text').innerText = timeLeft;
      const circleProgress = document.getElementById('timer-progress');
      const offset = fullDash - (timeLeft / 10) * fullDash;
      circleProgress.style.strokeDashoffset = offset;

      if (timeLeft <= 3) {
        circleProgress.style.stroke = '#e74c3c';
      } else {
        circleProgress.style.stroke = '#2ecc71';
      }
    }

    function showQuestion() {
      canAnswer = true;
      timeLeft = 10;
      updateTimerUI();

      const qData = selectedQuestions[currentQuestionIndex];
      document.getElementById('progress-text').innerText = `السؤال ${currentQuestionIndex + 1} من ${selectedQuestions.length}`;
      document.getElementById('question-text').innerText = qData.q;

      const optionsContainer = document.getElementById('options-container');
      optionsContainer.innerHTML = '';

      qData.options.forEach((opt, index) => {
        const btn = document.createElement('button');
        btn.className = 'btn-option';
        btn.innerText = opt;
        btn.onclick = () => checkAnswer(index, btn);
        optionsContainer.appendChild(btn);
      });

      clearInterval(timerInterval);
      timerInterval = setInterval(() => {
        timeLeft--;
        updateTimerUI();

        if (timeLeft <= 0) {
          clearInterval(timerInterval);
          handleTimeOut();
        }
      }, 1000);
    }

    function checkAnswer(selectedIndex, btnElement) {
      if (!canAnswer) return;
      canAnswer = false;
      clearInterval(timerInterval);

      const correctIndex = selectedQuestions[currentQuestionIndex].answer;
      const allButtons = document.querySelectorAll('.btn-option');

      if (selectedIndex === correctIndex) {
        btnElement.classList.add('correct');
        totalScore += timeLeft;
        document.getElementById('score-live').innerText = `النقاط: ${totalScore}`;
      } else {
        btnElement.classList.add('wrong');
        allButtons[correctIndex].classList.add('correct');
      }

      nextQuestionWithDelay();
    }

    function handleTimeOut() {
      canAnswer = false;
      const correctIndex = selectedQuestions[currentQuestionIndex].answer;
      const allButtons = document.querySelectorAll('.btn-option');
      allButtons[correctIndex].classList.add('correct');

      nextQuestionWithDelay();
    }

    function nextQuestionWithDelay() {
      setTimeout(() => {
        currentQuestionIndex++;
        if (currentQuestionIndex < selectedQuestions.length) {
          showQuestion();
        } else {
          showResults();
        }
      }, 1200);
    }

    function showResults() {
      clearInterval(timerInterval);
      document.getElementById('quiz-screen').style.display = 'none';
      document.getElementById('result-screen').style.display = 'block';
      document.getElementById('final-score').innerText = `مجموع نقاطك: ${totalScore}`;
    }

    loadQuestions();
  </script>
